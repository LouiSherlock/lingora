## 整体流程

```
点击「登录」→ 弹出 6 位 OTP 输入框 → 填满 6 位自动提交
  → POST /api/auth { code }
  → 后端比对写死的 code → 匹配则签发 JWT（1 小时）
  → 前端保存 token
  → 之后所有 /api/* 请求带 Authorization: Bearer <token>
  → proxy.ts 统一校验，缺失、无效或过期都返回 401
  → 前端收到 401 后清除 token，重新弹出登录框
```

## 步骤 2：JWT 工具 `lib/jwt.ts`

```ts
import { SignJWT, jwtVerify } from 'jose'

const raw = process.env.AUTH_SECRET
if (!raw) throw new Error('AUTH_SECRET is not set')
const secret = new TextEncoder().encode(raw)

export function signToken() {
  return new SignJWT({ role: 'user' })
    .setProtectedHeader({ alg: 'HS256' })
    .setSubject('login-user')
    .setIssuedAt()
    .setExpirationTime('1h') // 有效期 1 小时
    .sign(secret)
}

export function verifyToken(token: string) {
  return jwtVerify(token, secret, { algorithms: ['HS256'] })
}
```

## 步骤 3：登录接口 `app/api/auth/route.ts`

```ts
import { NextResponse } from 'next/server'
import { timingSafeEqual } from 'node:crypto'
import { signToken } from '@/lib/jwt'

const VALID_CODE = process.env.AUTH_CODE ?? '123456'

function safeEqual(a: string, b: string) {
  const x = Buffer.from(a)
  const y = Buffer.from(b)
  return x.length === y.length && timingSafeEqual(x, y)
}

export async function POST(req: Request) {
  const body = await req.json().catch(() => null)
  const code = typeof body?.code === 'string' ? body.code : ''

  if (!/^\d{6}$/.test(code)) {
    return NextResponse.json({ message: '请输入 6 位数字验证码' }, { status: 400 })
  }
  if (!safeEqual(code, VALID_CODE)) {
    return NextResponse.json({ message: '验证码错误' }, { status: 401 })
  }

  const token = await signToken()
  return NextResponse.json({ token, expiresIn: 3600 })
}
```

## 步骤 4：统一鉴权 `proxy.ts`（放在项目根目录，或 `src/` 下）

```ts
import { NextRequest, NextResponse } from 'next/server'
import { errors } from 'jose'
import { verifyToken } from '@/lib/jwt'

const PUBLIC_PATHS = new Set(['/api/auth'])

function unauthorized(code: string, message: string) {
  return NextResponse.json(
    { code, message },
    { status: 401, headers: { 'WWW-Authenticate': 'Bearer' } },
  )
}

export async function proxy(req: NextRequest) {
  if (PUBLIC_PATHS.has(req.nextUrl.pathname)) return NextResponse.next()

  const auth = req.headers.get('authorization')
  const token = auth?.startsWith('Bearer ') ? auth.slice(7) : null
  if (!token) return unauthorized('MISSING_TOKEN', '未登录')

  try {
    await verifyToken(token)
    return NextResponse.next()
  } catch (e) {
    if (e instanceof errors.JWTExpired) {
      return unauthorized('TOKEN_EXPIRED', '登录已过期')
    }
    return unauthorized('INVALID_TOKEN', '无效的 token')
  }
}

export const config = {
  matcher: ['/api/:path*'], // 只拦截 API，页面不受影响
}
```

以后新增的接口只要放在 `/api/*` 下，就会自动受保护，不用每个接口单独加校验。

示例受保护接口 `app/api/profile/route.ts`：

## 步骤 5：前端

### 5.1 `lib/auth-client.ts`：token 存取和 fetch 封装

```ts
'use client'
import { useEffect, useState } from 'react'

const KEY = 'access_token'

export const getToken = () =>
  typeof window === 'undefined' ? null : localStorage.getItem(KEY)

export function setToken(t: string) {
  localStorage.setItem(KEY, t)
  window.dispatchEvent(new Event('auth:change'))
}

export function clearToken() {
  localStorage.removeItem(KEY)
  window.dispatchEvent(new Event('auth:change'))
}

// 所有业务请求都用它，自动带 token，遇到 401 就清 token 并通知弹登录框
export async function apiFetch(input: RequestInfo | URL, init: RequestInit = {}) {
  const headers = new Headers(init.headers)
  const token = getToken()
  if (token) headers.set('Authorization', `Bearer ${token}`)

  const res = await fetch(input, { ...init, headers })
  if (res.status === 401) {
    clearToken()
    window.dispatchEvent(new Event('auth:expired'))
  }
  return res
}

export function useAuth() {
  const [authed, setAuthed] = useState(false)
  useEffect(() => {
    const sync = () => setAuthed(!!getToken())
    sync()
    window.addEventListener('auth:change', sync)
    return () => window.removeEventListener('auth:change', sync)
  }, [])
  return authed
}
```

### 5.2 `components/OtpInput.tsx`

用一个隐藏的真实 input 加 6 个展示格子。这样粘贴、移动端数字键盘和短信验证码自动填充都能正常工作，不需要处理 6 个 input 之间的焦点跳转。

```tsx
'use client'
import { useRef } from 'react'

type Props = {
  value: string
  onChange: (v: string) => void
  onComplete?: (v: string) => void
  length?: number
  disabled?: boolean
}

export function OtpInput({ value, onChange, onComplete, length = 6, disabled }: Props) {
  const ref = useRef<HTMLInputElement>(null)

  return (
    <div className="relative flex gap-2" onClick={() => ref.current?.focus()}>
      <input
        ref={ref}
        autoFocus
        inputMode="numeric"
        autoComplete="one-time-code"
        maxLength={length}
        value={value}
        disabled={disabled}
        onChange={(e) => {
          const v = e.target.value.replace(/\D/g, '').slice(0, length)
          onChange(v)
          if (v.length === length) onComplete?.(v)
        }}
        className="absolute inset-0 z-10 opacity-0"
      />
      {Array.from({ length }).map((_, i) => (
        <div
          key={i}
          className={`flex h-12 w-10 items-center justify-center rounded-md border text-xl
            ${i === value.length && !disabled ? 'border-blue-500 ring-2 ring-blue-200' : 'border-gray-300'}`}
        >
          {value[i] ?? ''}
        </div>
      ))}
    </div>
  )
}
```

### 5.3 `components/LoginButton.tsx`：按钮 + 弹窗 + 过期自动弹出

```tsx
'use client'
import { useEffect, useState } from 'react'
import { OtpInput } from './OtpInput'
import { clearToken, setToken, useAuth } from '@/lib/auth-client'

export function LoginButton() {
  const authed = useAuth()
  const [open, setOpen] = useState(false)
  const [code, setCode] = useState('')
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  // 任何接口返回 401 后，自动弹出登录框
  useEffect(() => {
    const h = () => { setError('登录已过期，请重新登录'); setOpen(true) }
    window.addEventListener('auth:expired', h)
    return () => window.removeEventListener('auth:expired', h)
  }, [])

  async function submit(c: string) {
    setLoading(true)
    setError('')
    try {
      const res = await fetch('/api/auth', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ code: c }),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data.message ?? '登录失败')
      setToken(data.token)
      setOpen(false)
    } catch (e) {
      setError(e instanceof Error ? e.message : '登录失败')
    } finally {
      setCode('')
      setLoading(false)
    }
  }

  if (authed) {
    return <button onClick={clearToken} className="rounded border px-4 py-2">退出登录</button>
  }

  return (
    <>
      <button onClick={() => { setError(''); setOpen(true) }}
        className="rounded bg-black px-4 py-2 text-white">
        登录
      </button>

      {open && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40"
          onClick={() => !loading && setOpen(false)}>
          <div className="w-80 rounded-xl bg-white p-6 shadow-xl" onClick={(e) => e.stopPropagation()}>
            <h2 className="mb-4 text-lg font-medium">请输入 6 位验证码</h2>
            <OtpInput value={code} onChange={setCode} onComplete={submit} disabled={loading} />
            <p className="mt-3 h-5 text-sm text-red-500">{error}</p>
          </div>
        </div>
      )}
    </>
  )
}
```

## 步骤 6：验证

```bash
# 1. 未带 token，应返回 401
curl -i localhost:3000/api/profile

# 2. 错误 code，应返回 401
curl -i -X POST localhost:3000/api/auth -H 'Content-Type: application/json' -d '{"code":"000000"}'

# 3. 正确 code，拿到 token
TOKEN=$(curl -s -X POST localhost:3000/api/auth -H 'Content-Type: application/json' \
  -d '{"code":"123456"}' | jq -r .token)

# 4. 带 token，应返回 200
curl -i localhost:3000/api/profile -H "Authorization: Bearer $TOKEN"
```

**测试过期**：不用真等 1 小时。临时把 `setExpirationTime('1h')` 改成 `'10s'`，登录后等 10 秒再请求，应返回 `401 TOKEN_EXPIRED`，前端会自动弹出登录框。测完记得改回去。

## 几个需要注意的点

1. **6 位数字只有 100 万种组合，可以被暴力枚举。** 至少给 `/api/auth` 加失败限流，比如同一 IP 连续错 5 次锁 15 分钟。开发阶段可以用内存 Map，上线后换 Redis 或 Upstash。
2. **token 存储方式。** 上面用 localStorage + Bearer header，符合你"返回 token"的要求，实现也最简单，但 XSS 能读到它。要更安全的话，改成登录接口里 `Set-Cookie`（`httpOnly; Secure; SameSite=Lax; Max-Age=3600`），proxy 从 `req.cookies` 读取，前端就不用管 token 了。
3. **proxy 不要作为唯一防线。** Next.js 曾经出过中间件被绕过的漏洞（CVE-2025-29927，已修复，请保持版本最新）。对敏感接口，可以在 route handler 里再调一次 `verifyToken` 做二次校验。
4. **版本差异。** Next.js 16 起文件名是 `proxy.ts`，导出函数名是 `proxy`；15 及以下是 `middleware.ts`，导出 `middleware`，其余写法一致。
5. **跨域调用。** 如果以后有别的域名调用这些接口，需要在 proxy 里先放行 `OPTIONS` 预检请求，否则会被当成未登录。
6. `AUTH_SECRET` 和 `AUTH_CODE` 不要提交到 git，部署时在平台上配置环境变量。
